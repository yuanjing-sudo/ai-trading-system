# Or run quant analysis directly
from quant import indicators, strategies, risk, backtest
import pandas as pd

# Load price data
prices = pd.read_csv("data/AAPL.csv", index_col="date", parse_dates=True")

# Generate signals
signals = strategies.dual_momentum_strategy(prices, lookback=126)

# Run backtest
engine = backtest.BacktestEngine(initial_capital=100000)
result = engine.run(prices, signals)

# View performance
print(result.summary())
print(result.metrics)
